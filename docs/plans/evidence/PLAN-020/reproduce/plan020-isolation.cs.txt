System.Collections.IEnumerator Run() {
    var flags=System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.NonPublic;
    var load=(System.Action<string>)System.Delegate.CreateDelegate(typeof(System.Action<string>),typeof(MobileGame.Foundation.Editor.Plan020Probe).GetMethod("Load",flags));
    var root=MobileGame.Foundation.Editor.Plan020Probe.Root;
    var data=MobileGame.Presentation.GameDataStore.Instance.Catalog;
    var codec=new MobileGame.Persistence.SessionSnapshotCodec(data.Definitions.Values,true);
    foreach(var mode in new[]{"laboratory-only","invasion-only"}) {
        var source=System.IO.File.ReadAllText(System.IO.Path.Combine(root,mode=="laboratory-only"?"fixture-2.json":"fixture-0.json"));
        var state=codec.Restore(codec.Deserialize(source));var lab=(MobileGame.Gameplay.LaboratoryManager)state.Runtime;
        if(mode=="laboratory-only") {
            var external=System.Linq.Enumerable.ToArray(System.Linq.Enumerable.Where(lab.Registry.Things,t=>t.Definition is MobileGame.Foundation.CreatureData c && c.Role==MobileGame.Foundation.CreatureRole.Invader));
            foreach(var t in external)lab.Registry.Remove(t);
        } else {
            for(int i=0;i<16;i++) {var t=lab.Registry.Create(data.Selected<MobileGame.Foundation.CreatureData>("invaderId"));if(!lab.Map.Commands.Place(t.Id,new UnityEngine.Vector2Int(1,1)).Succeeded)throw new System.InvalidOperationException("invader preparation");}
            if(!lab.Invasion.ReleaseAll())throw new System.InvalidOperationException("release preparation");
        }
        lab.ValidateState();lab.Invasion.ValidateState();
        var snapshot=codec.Serialize(codec.Capture(state,MobileGame.Foundation.SessionPhase.Playing));
        System.IO.File.WriteAllText(System.IO.Path.Combine(root,"fixture-"+mode+".json"),snapshot);
        load(snapshot);var view=UnityEngine.Object.FindFirstObjectByType<MobileGame.Presentation.LaboratoryView>();view.SetUiPreferences(true,false);
        MobileGame.Foundation.Editor.Plan020Probe.Resize(1280,720);
        var task=MobileGame.Presentation.SceneFlowManager.Instance.Localization.SelectLanguageAsync("ko");while(!task.IsCompleted)yield return null;if(task.IsFaulted)throw task.Exception;
        yield return new UnityEngine.WaitForSecondsRealtime(1);
        MobileGame.Foundation.Editor.Plan020Probe.Measure("isolated-"+mode);
        while(MobileGame.Foundation.Editor.Plan020Probe.Status.StartsWith("measuring"))yield return null;
    }
    MobileGame.Foundation.Editor.Plan020Probe.Write("isolation-complete",MobileGame.Foundation.Editor.Plan020Probe.Snapshot());
}
UnityEngine.Object.FindFirstObjectByType<MobileGame.Presentation.LaboratoryView>().StartCoroutine(Run());
return true;
