void Bench(string name,System.Action action,int count=120) {
    for(int i=0;i<20;i++)action();
    var times=new double[count];var bytes=new long[count];
    for(int i=0;i<count;i++){
        var b=System.GC.GetAllocatedBytesForCurrentThread();var t=System.Diagnostics.Stopwatch.GetTimestamp();action();
        times[i]=(System.Diagnostics.Stopwatch.GetTimestamp()-t)*1000d/System.Diagnostics.Stopwatch.Frequency;
        bytes[i]=System.GC.GetAllocatedBytesForCurrentThread()-b;
    }
    System.Array.Sort(times);System.Array.Sort(bytes);
    results.Add(new{name,iterations=count,medianMs=times[count/2],p95Ms=times[(int)(count*.95)],medianBytes=bytes[count/2],p95Bytes=bytes[(int)(count*.95)]});
}
